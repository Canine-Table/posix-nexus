        \Verb#([a-zA-Z]|[-_:.])+$#

        \Verb#([-_:.])+$#

    \item[\textbf{9.4}] \textbf{Skip regex (whitespace)}  \\
        \Verb#(\n\v\f\t\r )+#

    \item[\textbf{9.5}] \textbf{Action regex (OR of action chars)}  \\

        \Verb#^(\n\v\f\t\r )+$#

    \item[\textbf{9.7}] \textbf{Macro regex}  \\
        \Verb#<nx@([a-zA-Z]|[-_:.])+\/>#

    \item[\textbf{9.8}] \textbf{Macro remove‑prefix‑suffix regex}  \\
        \Verb#(^<nx@|\/>$)#
\end{enumerate}
