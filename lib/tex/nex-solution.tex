\def\sDigit#1{%
	\ifx#1\nx@empty%
		-1% invalid
	\else%
		\ifnum\number`#1<48
			-2%
		\else
			\ifnum\number`#1>57%
				-3%
			\else%
				\sNumExpr{\number`#1-48}% returns numeric digit 0–9
			\fi%
		\fi%
	\fi%
}

\def\sEmpty#1#2#3{%
	\edef\s@tmpa{\unexpanded{#1}}%
	\ifx\s@tmpa\s@empty%
		#2%
	\else%
		#3%
	\fi%
}

\def\sDefault#1#2#3{%
	\edef\s@tmpa{}%
	\def\s@tmpa{#1}%
	\ifx\s@tmpa\s@default%
		#2%
	\else%
		#3%
	\fi%
}

\def\sCs#1#2#3{%
	\expandafter\ifx\csname#1\endcsname\relax%
		#3%
	\else%
		#2%
	\fi%
}

\def\sRegister#1#2{%
	\sEmpty{#1}{%
		\typeout{empty name for a register is not allowed}%
	}{%
		\sCs{#1}{%
			\typeout{Looks like the register '#1' is already a household name...}
		}{%
			\edef\s@tmpb{\sDigit{#2}}
			\ifnum\s@tmpb>-1
				\ifcase\s@tmpb
					\let\s@tmpa=\newdimen% Case 0: Dimension register
				\or%
					\let\s@tmpa=\newcount% Case 1: Count register
				\or%
					\let\s@tmpa=\newskip% Case 2: Glue register (skip)
				\or%
					\let\s@tmpa=\newmuskip% Case 3: Math glue register (muskip)
				\or%
					\let\s@tmpa=\newbox % Case 4: Box register
				\or%
					\let\s@tmpa=\newinsert% Case 5: Insert register
				\else
					\let\s@tmpa=\newtoks% Case (Default): Token register
				\fi\global\expandafter\s@tmpa\csname#1\endcsname\relax%
			\fi
		}
	}
}

\long\def\sCoverPage[#1]{%
	\pgfkeys{%
		solution community cover page,%
		title,%
		author,%
		ref,%
		abstract,%
		Abstract,%
		image,%
		#1%
	}%
	\def\s@ref{\pgfkeysvalueof{/solution community/cover page/ref}}
	\begin{tikzpicture}[remember picture, overlay]
		\shade[
			right color=solution.primary.black,%
			left color=solution.primary.black!70!solution.primary.white,%
			draw=solution.secondary.blue,%
			ultra thick,%
		] (current page.north west) -| (current page.south east) -| (current page.north west);
		\node[%
			circle,%
			outer color=solution.primary.black,%
			inner color=solution.primary.white,%
			minimum width=0.7\pagewidth,%
			circular drop shadow={%
				shadow scale=1.05%
			},%
			draw=solution.primary.black,%
			ultra thick,%
			postaction={%
				decorate,%
				decoration={%
					raise=2mm,
					text along path,%
					text align={align=center},%
					text color=solution.secondary.blue,%
					text={\today}
				}
			}
		] (logo) at (current page.center) {};
		\node at (current page.center) {\includesvg[width=10cm]{\s@image}};
		\sShadowed[at={( 0,5 )},%
			above=of logo,%
			text=solution.secondary.blue,%
			yshift=1cm,%
			align=center,%
			text width=\linewidth,%
			yslant=0.1%
		]{\sSFont{24pt} \bf \s@title};
		\node[below=of logo, text=solution.secondary.blue, yshift=-1cm, align=center, text width=\linewidth] (author) {\sSFont{24pt} \bf \s@author};
		\node[below=of author, text=solution.secondary.blue, align=center, text width=\linewidth] (link) {\sSFont{24pt} \href{\s@link}{\bf \s@ref}};
	\end{tikzpicture}
	\newpage
	\expandafter\ifx\s@abstract\relax\else
		\begin{tcolorbox}[nxCol=light, sdwA]
			\begin{tcolorbox}[nxCol=light, title={\bf \s@Abstract}, hdrD, sdwA]
				\s@abstract
			\end{tcolorbox}
		\end{tcolorbox}
		\bigskip
	\fi
}
